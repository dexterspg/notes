# run container  
# --name , set container name
# -d  run in detached mode
# with images postgres, dpage/pgadmin4  (will be downloaded from docker hub, if not present locally)


docker run -d --name postgredb -e POSTGRES_PASSWORD=root postgres
docker run -d --name pgadmin -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=root dpage/pgadmin4

#remove containers

docker rm -f  <containername> <containername2>

##create network


docker network create <networkname>

#networkname is my-network as example

docker run -d --name postgredb --network my-network -e  POSTGRES_PASSWORD=root postgres
docker run -d --name pgadmin --network my-network -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=root dpage/pgadmin4

# ping check reachability
docker exec -it pgadmin ping postgredb
